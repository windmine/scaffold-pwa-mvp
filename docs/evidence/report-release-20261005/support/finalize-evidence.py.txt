def require(condition, message):
    if not condition:
        raise ValueError(message)


def sha(data):
    return hashlib.sha256(data).hexdigest()


def canonical(value):
    return sha(json.dumps(value, sort_keys=True, separators=(",", ":")).encode())


def stamp(value):
    require(isinstance(value, str) and re.search(r"(?:Z|[+-]\d\d:\d\d)$", value), "UTC timestamp required")
    return datetime.fromisoformat(value.replace("Z", "+00:00")).astimezone(timezone.utc)


def inside(path):
    path = path.resolve()
    require(path.is_relative_to(ROOT) and not path.is_symlink(), "repository file required")
    return path


def checked_count(proof, count, strings=False):
    checks = proof["checks"]
    require(isinstance(checks, list) and len(checks) == count, f"exactly {count} checks required")
    names = checks if strings else [item["name"] for item in checks]
    require(len(set(names)) == count, "duplicate checks refused")
    require(strings or all(item.get("status") == "passed" for item in checks), "every check must pass")


def mobile_log(text):
    text = re.sub(r"\x1b\[[0-?]*[ -/]*[@-~]", "", text).replace("\r\n", "\n")
    marker = "10 native browser startup/asynchronous/queue-wait contract groups passed."
    require(" check:mobile\n" in text and text.count(marker) == 1, "complete check:mobile log boundary required")
    require(not re.search(r"\bnot\s+ok\b", text, re.I), "not ok present in mobile log")
    require(text.rstrip().endswith("browser workflow checks passed"), "terminal browser success line required")
    names = re.findall(r"^ok - (.+)$", text.split(marker, 1)[1], re.M)
    require(len(names) == len(set(names)) == 72, "exactly 72 distinct real-backend workflow passes required")
    require("11 installer checks passed; no backend, cloud, or account mutations." in text, "installer11 marker missing")
    require("mobile/browser workflow checks passed" in text, "static mobile shell marker missing")
    return names


def assets(proof, count):
    rows = proof["assets"]
    require(len(rows) == count and len({item["path"] for item in rows}) == count, "asset count mismatch")
    files = {}
    for item in rows:
        file = inside(ROOT / "dist" / ("index.html" if item["path"] == "/" else item["path"].lstrip("/")))
        require(file.is_relative_to(ROOT / "dist"), "asset outside dist")
        data = file.read_bytes()
        require(sha(data) == item["sha256"] and len(data) == item["bytes"], "frozen dist asset mismatch")
        files[file] = sha(data)
    return files


def channel(proof, name):
    return proof["channels"][name] if "channels" in proof else proof


def production_rewrites(value):
    rows = value["rewrites"]
    require(len(rows) == 2, "exactly API and upload rewrites required")
    require({item.get("glob", item.get("source")) for item in rows} == {"/api/**", "/uploads/**"}, "rewrite scope mismatch")
    require(all(item["run"] == {"serviceId": "geo-backend", "region": "australia-southeast1"} for item in rows), "nonproduction rewrite refused")

    def bind(path, digest):
        require(path not in inputs or inputs[path] == digest, "input changed during validation")
        inputs[path] = digest

        try:
            action()
            gates.append(label)
        except (ValueError, KeyError, TypeError, OSError, AttributeError) as error:
            blockers.append({"gate": label, "reason": str(error) if isinstance(error, ValueError) else type(error).__name__})

    def load(key):

        data = path.read_bytes()

        bind(path, sha(data))
        proofs[key] = proof
        if key in ("pg17", "pg18"):

        else:
            expected = "SUCCESS" if key == "build" else "diagnosed_and_verifier_repaired" if key == "note_diagnosis" else "historical_verifier_failure_cause_unconfirmed" if key == "ux_boundary_diagnosis" else "passed"

    for key in paths:
        gate(f"proof:{key}", lambda key=key: load(key))

        require(proofs["plan"]["job"]["spec"]["template"]["spec"]["template"]["spec"]["containers"][0]["args"] == ["-m", "app.migrations", "--check"], "stage job must be check-only")
        require(proofs["preflight"]["migrationRequired"] is False and proofs["preflight"]["maintenanceRequired"] is False, "no migration/maintenance requirement proof missing")
    gate("source-build-stage-bindings", bindings)

    def preservation():

                c = p["cleanup"]
                require(c["templateArchived"] is True, "owned Template not archived")

                    else:
                        require(c["submissionOutcomeUnknown"] is False and p["photoCount"] == 50 and p["assertNonblockingStartup"] is True and p["retainedWorkerBaselinesVerified"] is True, "photo50 scope/outcome/history gate missing")
                        require(p["partialUploadCheckpoint"]["uploadedPhotos"] == 25 and p["partialUploadCheckpoint"]["originalPhotoHashesVerified"] == 50, "25+25 original evidence checkpoint missing")

    gate("old-to-new-waiting-update", waiting)

    def final_infrastructure():

        require(p["phase"] == "live" and p["sourceCommit"] == SOURCE and p["image"] == IMAGE and p["buildId"] == BUILD and p["revision"] == REVISION and p["previousRevision"] == PREVIOUS, "final infrastructure identity mismatch")

        recovered = proofs["recovery"]
        require(recovered["projectId"] == "plain-smoke-06194407" and recovered["productionBranch"]["id"] == "br-snowy-tree-a7hfuph6" and recovered["releasePhase"] == "Current", "Current production recovery required")

        require(all(stamp(h["recordedAtUtc"]) >= stamp(proofs[f"live_{kind}"]["completedAtUtc"]) for kind in ("shell", "installer", "october", "ux", "photo50")), "final hardening must follow all live checks")
    gate("final-infrastructure-recovery-hardening", final_infrastructure)

            require(len(p["pdfs"]) == 2 and sum(item["pages"] for item in p["pdfs"]) == 18, "18 PDF pages required")
            for item in p["pdfs"]:
                file = inside(ROOT / item["path"]); data = file.read_bytes(); inputs[file] = sha(data)
                require(inputs[file] == item["sha256"] and item["mutualLogoVerifiedEveryPage"] is True and item["eachPhotoAndSignaturePixelsVerified"] is True, "PDF artifact/evidence hash mismatch")
        parity = proofs["pdf_parity"]
        require(parity["pages"] == parity["pixelIdenticalPages"] == len(parity["checks"]) == 18 and all(item["identical"] is True for item in parity["checks"]), "18 identical candidate/live rendered pages required")
        visual = proofs["pdf_visual"]
        require(visual.get("reviewedPages", visual.get("totalPagesVisuallyReviewed")) == 18 and visual.get("physicalPhonePdfViewerVerified", False) is False, "18-page visual review required; no phone certification")
    gate("presentation-pdf-preservation-and-pixel-parity", pdfs)

    def cleanup():
        p, n = proofs["teardown"], proofs["neon_cleanup"]
        require(p["runId"] == "20261005" and p["deletionComplete"] is True and p["productionUnchanged"] is True and p["uploadsUntouched"] is True and p["neonUntouched"] is True, "exact GCP cleanup/upload-retention proof required")
        for key, proof_key in (("snapshot", "snapshot"), ("resources", "resources"), ("service", "service"), ("stagePlan", "plan")):
            require(p["creationProofSha256"][key] == canonical(proofs[proof_key]), "teardown creation proof binding mismatch")
        require({item["action"] for item in p["attempts"]} == {f"delete:{name}" for name in ("geo-report-stage-20261005", "geo-report-migrate-stage-20261005", "report-stage-db-20261005", "report-stage-jwt-20261005")} and len(p["attempts"]) == 4, "exact four GCP deletes required")
        require(n["branchId"] == "br-green-mode-a7x9jzqz" and n["projectId"] == "plain-smoke-06194407" and n["parentId"] == "br-snowy-tree-a7hfuph6" and all(n[k] is True for k in ("exactOwnershipVerified", "gcpStageResourcesAlreadyDeleted", "deleted", "absenceVerified", "productionAndBackupRetained")), "exact Neon cleanup and retention required")
