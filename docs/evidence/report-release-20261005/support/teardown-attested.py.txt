"""One-off exact October 5 staging attestation; no blanket clock tolerance.

inspect is read-only and emits sanitized evidence for review. plan is read-only.
delete requires --allow-stage-deletion and the passed attested plan. Secret
payloads, Neon, uploads and production configuration are never mutated here.
The original repository teardown implementation and chronology guard stay intact.
"""
import argparse
import copy
import hashlib
import importlib.util
import json
from pathlib import Path
import re

ROOT = Path(__file__).resolve().parents[2]
SPEC = importlib.util.spec_from_file_location("october_stage_teardown", ROOT / "scripts/teardown-report-release-stage.py")
teardown = importlib.util.module_from_spec(SPEC)
SPEC.loader.exec_module(teardown)
operator = teardown.operator
require = operator.require
RUN = "20261005"
NUMBER = "794826041820"
NAMES = operator.names(RUN)
EVIDENCE = ROOT / "docs/evidence/report-release-20261005"
ATTESTATION = EVIDENCE / "stage-creation-attestation.json"
PLAN = EVIDENCE / "stage-teardown-plan-attested.json"
DELETION = EVIDENCE / "stage-teardown.json"
KINDS = ("service", "job", "databaseSecret", "jwtSecret")
ORIGINAL_WINDOW = teardown.creation_window
ORIGINAL_COMMAND = operator.command
CALLER_SHA = "31ab631247bfe79050ce33abe0a6dc6a75ee28bd8117a05506dcb94f2db71498"
INSERT_IDS = {"service": "15cgu7hdc3og", "job": "2st57ddc5w7", "databaseSecret": "d1jk5re20loi", "jwtSecret": "1kx6bihe2bk1g"}
OBSERVED_UIDS = {"service": "01eb321a-3bb9-4680-832e-905b64a5b80b", "job": "2d234323-227b-4a00-8954-cf008ede5477"}
OBSERVED_CREATED = {"service": "2026-10-04T22:01:58.558097Z", "job": "2026-10-04T21:59:16.672374Z",
                    "databaseSecret": "2026-10-04T21:58:47.391981Z", "jwtSecret": "2026-10-04T21:59:01.537146Z"}
PROOF_FILES = {"snapshot": "snapshot.json", "resources": "stage-resources.json", "service": "stage-service.json", "stagePlan": "stage-plan.json"}


def bounded_command(executable, args, **kwargs):
    kwargs["timeout"] = min(kwargs.get("timeout", 90), 90)
    return ORIGINAL_COMMAND(executable, args, **kwargs)


operator.command = bounded_command


def file_sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def action_for(kind):
    return f"create_{kind if kind in ('service', 'job') else 'secret'}:{NAMES[kind]}"


def source_proofs():
    values = {key: json.loads((EVIDENCE / filename).read_text(encoding="utf-8")) for key, filename in PROOF_FILES.items()}
    teardown.validate_proofs(values["snapshot"], values["resources"], values["service"], run_id=RUN)
    teardown.verify_autogenerated_stage_plan(values["stagePlan"], values["snapshot"], values["resources"], values["service"], run_id=RUN)
    require(teardown.migration_mode(values["resources"]) == "none", "check_only_creation_required")
    return values


def source_bindings(values):
    return {key: {"file": filename, "fileSha256": file_sha(EVIDENCE / filename), "sha256": operator.sha(values[key])}
            for key, filename in PROOF_FILES.items()}


def code_bindings():
    paths = {"wrapper": Path(__file__), "teardown": ROOT / "scripts/teardown-report-release-stage.py",
             "operator": ROOT / "scripts/report-release-operator.py"}
    return {key: file_sha(path) for key, path in paths.items()}


def fresh_resources():
    project = operator.cloud("projects", "describe", operator.PROJECT)
    require(project.get("projectId") == operator.PROJECT and str(project.get("projectNumber")) == NUMBER, "project_identity_mismatch")
    current = {"service": teardown.read_service(RUN), "job": teardown.read_job(RUN)}
    for kind in ("databaseSecret", "jwtSecret"):
        current[kind] = teardown.read_secret_metadata(NAMES[kind], RUN)
    return current


def resource_binding(kind, value):
    run_object = kind in ("service", "job")
    metadata = value.get("metadata", {}) if run_object else value.get("secret", {})
    name = metadata.get("name")
    created = metadata.get("creationTimestamp") if run_object else metadata.get("createTime")
    require(name == (NAMES[kind] if run_object else f"projects/{NUMBER}/secrets/{NAMES[kind]}")
            and created == OBSERVED_CREATED[kind], "attested_resource_identity_mismatch")
    if run_object:
        require(metadata.get("uid") == OBSERVED_UIDS[kind] and str(metadata.get("namespace")) == NUMBER,
                "attested_resource_uid_mismatch")
    return {"name": name, "uid": metadata.get("uid") if run_object else None,
            "createdAtUtc": created, "resourceSha256": operator.sha(value),
            "configurationSha256": operator.sha(value["spec"] if run_object else {
                "labels": metadata.get("labels"), "replication": metadata.get("replication"),
                "versions": value["versions"], "policy": value["policy"]})}


def read_lifecycle_events():
    # Exact names, no broad project teardown. A complete bounded interval also
    # excludes delete/recreate between original creation and this fresh read.
    resources = []
    for project in (operator.PROJECT, NUMBER):
        resources.extend([f"namespaces/{project}/services/{NAMES['service']}",
                          f"namespaces/{project}/jobs/{NAMES['job']}",
                          f"projects/{project}/secrets/{NAMES['databaseSecret']}",
                          f"projects/{project}/secrets/{NAMES['jwtSecret']}"])
    until = operator.now()
    query = (f'logName="projects/{operator.PROJECT}/logs/cloudaudit.googleapis.com%2Factivity" '
             f'AND timestamp>="2026-10-04T21:58:00Z" AND timestamp<="{until}" AND ('
             + " OR ".join(f'protoPayload.resourceName="{name}"' for name in resources) + ")")
    events = operator.cloud("logging", "read", query, "--limit=500", "--order=asc")
    require(isinstance(events, list) and len(events) < 500, "lifecycle_audit_incomplete")
    account = operator.command("gcloud", ["config", "get-value", "account"], json_output=False)
    require(hashlib.sha256(account.encode()).hexdigest() == CALLER_SHA, "active_account_changed")
    return events, until


def request_configuration(kind, request, values):
    """Validate all available audit fields; never pretend omitted fields exist."""
    run_object = kind in ("service", "job")
    key = kind if run_object else "secret"
    require(set(request) == ({"@type", "parent", key} if run_object else {"@type", "parent", "secret", "secretId"}),
            "audit_request_shape_changed")
    require(request["parent"] == (f"namespaces/{operator.PROJECT}" if run_object else f"projects/{operator.PROJECT}"),
            "audit_request_parent_mismatch")
    actual = copy.deepcopy(request[key])
    baseline, resources = values["snapshot"], values["resources"]
    if run_object:
        expected = operator.service_document(baseline, NAMES, resources["artifact"]["image"]) if kind == "service" else operator.job_document(
            baseline, NAMES, resources["artifact"]["image"], migration_mode="none")
        require(actual["metadata"].pop("namespace", None) == operator.PROJECT, "audit_namespace_mismatch")
        if kind == "job":
            annotations = {"run.googleapis.com/client-name": "gcloud", "run.googleapis.com/client-version": "571.0.0"}
            require(actual["metadata"].pop("annotations", None) == annotations
                    and actual["spec"]["template"].pop("metadata", None) == {"annotations": annotations},
                    "audit_generated_metadata_changed")
            container = expected["spec"]["template"]["spec"]["template"]["spec"]["containers"][0]
            omitted = ["command", "args", "env"]
        else:
            container = expected["spec"]["template"]["spec"]["containers"][0]
            omitted = ["env", "startupProbe", "ports"]
        for field in omitted:
            require(field in container, "expected_audit_omission_missing")
            container.pop(field)
    else:
        require(request["secretId"] == NAMES[kind], "audit_secret_name_mismatch")
        expected = {"replication": {"automatic": {}}}
        omitted = ["labels"]
    require(actual == expected, "audit_available_configuration_mismatch")
    return {"requestSha256": operator.sha(request), "availableConfigurationSha256": operator.sha(expected),
            "omittedCreationFields": omitted, "availableConfigurationMatchesPlan": True}


def audit_binding(kind, events, binding, values):
    names = ({f"namespaces/{project}/{'services' if kind == 'service' else 'jobs'}/{NAMES[kind]}" for project in (operator.PROJECT, NUMBER)}
             if kind in ("service", "job") else {f"projects/{project}/secrets/{NAMES[kind]}" for project in (operator.PROJECT, NUMBER)})
    matching = [event for event in events if event.get("protoPayload", {}).get("resourceName") in names]
    method = ("google.cloud.run.v1.Services.CreateService" if kind == "service" else
              "google.cloud.run.v1.Jobs.CreateJob" if kind == "job" else "google.cloud.secretmanager.v1.SecretManagerService.CreateSecret")
    creates = [event for event in matching if event.get("protoPayload", {}).get("methodName", "").rsplit(".", 1)[-1].startswith("Create")]
    require(len(creates) == 1, "unique_creation_event_required")
    require(not any(event.get("protoPayload", {}).get("methodName", "").rsplit(".", 1)[-1].startswith(("Delete", "Replace"))
                    for event in matching), "intervening_resource_lifecycle_refused")
    event = creates[0]
    payload = event["protoPayload"]
    require(payload.get("methodName") == method and payload.get("status", {}).get("code", 0) == 0
            and event.get("insertId") == INSERT_IDS[kind] and not event.get("operation"), "exact_successful_creation_event_required")
    require(payload.get("serviceName") == ("run.googleapis.com" if kind in ("service", "job") else "secretmanager.googleapis.com"),
            "audit_service_mismatch")
    caller = payload.get("authenticationInfo", {}).get("principalEmail", "")
    require(hashlib.sha256(caller.encode()).hexdigest() == CALLER_SHA
            and "gcloud" in payload.get("requestMetadata", {}).get("callerSuppliedUserAgent", ""), "audit_caller_mismatch")
    # These are provider-side times only, not a guessed local-clock offset.
    require(teardown.utc(event["timestamp"]) <= teardown.utc(binding["createdAtUtc"]) <= teardown.utc(event["receiveTimestamp"]),
            "provider_creation_outside_exact_event")
    config = request_configuration(kind, payload.get("request", {}), values)
    if kind not in ("service", "job"):
        response = payload.get("response", {})
        require(response.get("name") == binding["name"] and response.get("createTime") == binding["createdAtUtc"],
                "audit_secret_response_identity_mismatch")
    proof = values["service"] if kind == "service" else values["resources"]
    require(teardown.utc(proof["startedAtUtc"]) <= teardown.utc(event["timestamp"])
            <= teardown.utc(event["receiveTimestamp"]) <= teardown.utc(proof["finishedAtUtc"]), "audit_outside_owned_phase")
    return {"insertId": event["insertId"], "timestamp": event["timestamp"], "receiveTimestamp": event["receiveTimestamp"],
            "methodName": method, "resourceName": payload["resourceName"], "callerSha256": CALLER_SHA,
            "statusCode": 0, "eventSha256": operator.sha(event), "uniqueCreateAndNoDeleteOrReplace": True,
            "providerCreationWithinEvent": True, **config}


def exact_window(proof, action, created, entries):
    try:
        return ORIGINAL_WINDOW(proof, action, created)
    except RuntimeError as error:
        require(str(error) == "resource_not_created_during_owned_attempt", "non_chronology_override_refused")
    matches = [entry for entry in entries.values() if entry["action"] == action and entry["creationProofSha256"] == operator.sha(proof)
               and entry["resource"]["createdAtUtc"] == created]
    require(len(matches) == 1 and proof.get("runId") == RUN and proof.get("resources") == NAMES, "unattested_creation_refused")
    attempts = [item for item in proof.get("attempts", []) if item.get("action") == action]
    require(len(attempts) == 1 and teardown.utc(proof["startedAtUtc"]) <= teardown.utc(created)
            < teardown.utc(attempts[0]["atUtc"]) <= teardown.utc(proof["finishedAtUtc"]), "invalid_owned_chronology")


def install_verified_window(entries):
    teardown.creation_window = lambda proof, action, created: exact_window(proof, action, created, entries)


def verify_existing_guards(current, values, entries):
    install_verified_window(entries)
    try:
        teardown.verify_service(current["service"], values["snapshot"], values["resources"], values["service"], NUMBER,
                                stage_plan=values["stagePlan"], run_id=RUN)
        teardown.verify_job(current["job"], values["snapshot"], values["resources"], NUMBER, run_id=RUN)
        teardown.require_no_running_job(RUN)
        for kind in ("databaseSecret", "jwtSecret"):
            value = current[kind]
            teardown.verify_secret(value["secret"], value["versions"], value["policy"], NAMES[kind], values["resources"], NUMBER, run_id=RUN)
    finally:
        teardown.creation_window = ORIGINAL_WINDOW


def collect_attestation():
    values = source_proofs()
    current = fresh_resources()
    events, until = read_lifecycle_events()
    entries = {}
    for kind in KINDS:
        binding = resource_binding(kind, current[kind])
        proof_key = "service" if kind == "service" else "resources"
        entries[kind] = {"action": action_for(kind), "creationProofSha256": operator.sha(values[proof_key]),
                         "resource": binding, "audit": audit_binding(kind, events, binding, values)}
        exact_window(values[proof_key], action_for(kind), binding["createdAtUtc"], entries)
    verify_existing_guards(current, values, entries)
    return {"schemaVersion": 1, "phase": "stage-creation-attestation", "status": "passed", "runId": RUN,
            "recordedAtUtc": operator.now(), "projectId": operator.PROJECT, "projectNumber": NUMBER,
            "resources": NAMES, "sourceProofs": source_bindings(values), "codeSha256": code_bindings(), "exceptions": entries,
            "lifecycleAuditThroughUtc": until, "secretPayloadReads": 0, "mutationCalls": 0,
            "scope": "Four exact create timestamps only; secret-version windows and all original identity/configuration/live guards remain strict.",
            "limitations": ["Cloud Run creation audit response has no UID; current exact UID and provider creation time within the unique event interval plus no intervening create/delete/replace bind the incarnation.",
                            "Audit requests omit the listed fields. Available fields match the original plan; complete current runtime, secret references, job arguments, policies and versions pass the original guards."]}


def attestation_comparable(value):
    return {key: item for key, item in value.items() if key not in ("recordedAtUtc", "lifecycleAuditThroughUtc")}


def verified_attestation():
    saved = operator.read_proof(ATTESTATION, "stage-creation-attestation")
    fresh = collect_attestation()
    require(attestation_comparable(fresh) == attestation_comparable(saved), "fresh_attestation_mismatch")
    require(teardown.utc(fresh["lifecycleAuditThroughUtc"]) >= teardown.utc(saved["lifecycleAuditThroughUtc"]), "audit_interval_regressed")
    return saved, fresh


def run_teardown(delete):
    values = source_proofs()
    require(ATTESTATION.is_file(), "creation_attestation_required")
    attestation_file_hash = file_sha(ATTESTATION)
    saved = operator.read_proof(ATTESTATION, "stage-creation-attestation")
    plan = operator.read_proof(PLAN, "stage-teardown-plan") if delete else None
    if delete:
        require(plan.get("runId") == RUN and plan.get("attestationFileSha256") == attestation_file_hash
                and plan.get("attestationSha256") == operator.sha(saved) and plan.get("deletionComplete") is False
                and plan.get("attempts") == [], "passed_exact_attested_plan_required")
    path = DELETION if delete else PLAN
    require(not path.exists(), "new_evidence_required")
    proof = {"schemaVersion": 1, "phase": "stage-teardown" if delete else "stage-teardown-plan", "status": "started",
             "runId": RUN, "resources": NAMES, "migrationMode": "none", "startedAtUtc": operator.now(), "attempts": [],
             "creationProofSha256": {key: operator.sha(value) for key, value in values.items()},
             "attestationFileSha256": attestation_file_hash, "attestationSha256": operator.sha(saved),
             "codeSha256": code_bindings(), "neonUntouched": True, "uploadsUntouched": True}
    if delete:
        proof["attestedPlanSha256"] = operator.sha(plan)
        proof["attestedPlanFileSha256"] = file_sha(PLAN)
    with path.open("x", encoding="utf-8") as file:
        json.dump(proof, file)
    def save():
        path.write_text(json.dumps(proof, indent=2) + "\n", encoding="utf-8")
    try:
        live = {"live": operator.live_snapshot()}
        saved, fresh = verified_attestation()
        install_verified_window(saved["exceptions"])
        owned = teardown.preflight(values["snapshot"], values["resources"], values["service"], stage_plan=values["stagePlan"], run_id=RUN)
        require({key: resource_binding(key, value) for key, value in owned.items()}
                == {key: entry["resource"] for key, entry in saved["exceptions"].items()}, "resource_changed_since_attestation")
        require(file_sha(ATTESTATION) == attestation_file_hash and source_bindings(source_proofs()) == saved["sourceProofs"], "proof_changed_after_preflight")
        operator.same_live(live)
        proof.update(verifiedResourceSha256={key: operator.sha(value) for key, value in owned.items()},
                     freshLifecycleAuditThroughUtc=fresh["lifecycleAuditThroughUtc"],
                     exactCreationExceptions=[entry["action"] for entry in saved["exceptions"].values()])
        save()
        if delete:
            require(proof["verifiedResourceSha256"] == plan.get("verifiedResourceSha256"), "resource_changed_since_attested_plan")
            for kind in KINDS:
                operator.same_live(live)
                current = teardown.read_service(RUN) if kind == "service" else teardown.read_job(RUN) if kind == "job" else teardown.read_secret_metadata(NAMES[kind], RUN)
                require(current == owned[kind], "stage_resource_changed_after_preflight")
                if kind == "job":
                    teardown.require_no_running_job(RUN)
                proof["attempts"].append({"action": f"delete:{NAMES[kind]}", "atUtc": operator.now()})
                save()
                teardown.delete_exact(kind, RUN)
            teardown.confirm_absent(RUN)
        operator.same_live(live)
        proof.update(status="passed", productionUnchanged=True, deletionComplete=delete, finishedAtUtc=operator.now())
        save()
        print(json.dumps({"status": "passed", "phase": proof["phase"], "deletedResources": len(proof["attempts"]), "evidence": str(path)}))
        return 0
    except Exception as error:
        code = str(error) if isinstance(error, RuntimeError) and re.fullmatch(r"[a-z_]+", str(error)) else "attested_teardown_failed_details_withheld"
        proof.update(status="failed", failureCode=code, finishedAtUtc=operator.now())
        save()
        print(json.dumps({"status": "failed", "code": code, "attempts": len(proof["attempts"]), "evidence": str(path)}))
        return 1
    finally:
        teardown.creation_window = ORIGINAL_WINDOW


def self_test():
    from unittest.mock import patch
    count = 0
    def check(condition):
        nonlocal count
        require(condition, "offline_test_failed")
        count += 1
    def rejects(callback):
        try:
            callback()
        except (RuntimeError, KeyError, ValueError):
            check(True)
        else:
            check(False)
    proof = {"runId": RUN, "resources": NAMES, "startedAtUtc": "2026-10-04T22:01:44Z", "finishedAtUtc": "2026-10-04T22:02:47Z",
             "attempts": [{"action": action_for("service"), "atUtc": "2026-10-04T22:01:58.803179Z"}]}
    entry = {"action": action_for("service"), "creationProofSha256": operator.sha(proof), "resource": {"createdAtUtc": OBSERVED_CREATED["service"]}}
    entries = {"service": entry}
    rejects(lambda: ORIGINAL_WINDOW(proof, entry["action"], OBSERVED_CREATED["service"]))
    exact_window(proof, entry["action"], OBSERVED_CREATED["service"], entries); check(True)
    rejects(lambda: exact_window(proof, entry["action"], OBSERVED_CREATED["service"], {}))
    rejects(lambda: exact_window(proof, entry["action"], "2026-10-04T22:01:58.557097Z", entries))
    rejects(lambda: exact_window({**proof, "runId": "other"}, entry["action"], OBSERVED_CREATED["service"], entries))
    rejects(lambda: exact_window({**proof, "attempts": proof["attempts"] * 2}, entry["action"], OBSERVED_CREATED["service"], entries))
    rejects(lambda: exact_window(proof, "add_secret_version:" + NAMES["databaseSecret"], OBSERVED_CREATED["databaseSecret"], entries))
    rejects(lambda: exact_window(proof, entry["action"], "2026-10-04T22:02:48Z", entries))
    rejects(lambda: exact_window(proof, entry["action"], "2026-10-04T22:01:43Z", entries))
    value = {"metadata": {"name": NAMES["service"], "uid": OBSERVED_UIDS["service"], "namespace": NUMBER,
                           "creationTimestamp": OBSERVED_CREATED["service"]}, "spec": {"mock": "configuration"}}
    check(resource_binding("service", value)["uid"] == OBSERVED_UIDS["service"])
    for field, wrong in (("uid", "wrong"), ("name", "geo-backend"), ("creationTimestamp", "2026-10-04T22:01:59Z")):
        changed = copy.deepcopy(value); changed["metadata"][field] = wrong
        rejects(lambda changed=changed: resource_binding("service", changed))
    changed = copy.deepcopy(value); changed["spec"]["mock"] = "drift"
    check(resource_binding("service", changed) != resource_binding("service", value))
    with patch.object(operator, "cloud", side_effect=AssertionError("no_cloud_in_self_test")):
        rejects(lambda: audit_binding("service", [], resource_binding("service", value), {}))
        duplicates = [{"protoPayload": {"resourceName": f"namespaces/{operator.PROJECT}/services/{NAMES['service']}",
                                       "methodName": "google.cloud.run.v1.Services.CreateService"}}] * 2
        rejects(lambda: audit_binding("service", duplicates, resource_binding("service", value), {}))
        wrong_hash = copy.deepcopy(entries); wrong_hash["service"]["creationProofSha256"] = "0" * 64
        rejects(lambda: exact_window(proof, entry["action"], OBSERVED_CREATED["service"], wrong_hash))
        values = source_proofs()
        config = operator.service_document(values["snapshot"], NAMES, values["resources"]["artifact"]["image"])
        config["metadata"]["namespace"] = operator.PROJECT
        for field in ("env", "startupProbe", "ports"):
            config["spec"]["template"]["spec"]["containers"][0].pop(field)
        event = {"timestamp": "2026-10-04T22:01:58.191456Z", "receiveTimestamp": "2026-10-04T22:01:58.834494506Z",
                 "insertId": INSERT_IDS["service"], "protoPayload": {
                     "methodName": "google.cloud.run.v1.Services.CreateService", "serviceName": "run.googleapis.com",
                     "resourceName": f"namespaces/{operator.PROJECT}/services/{NAMES['service']}",
                     "authenticationInfo": {"principalEmail": "offline@example.invalid"},
                     "requestMetadata": {"callerSuppliedUserAgent": "gcloud"},
                     "request": {"@type": "offline-fixture", "parent": f"namespaces/{operator.PROJECT}", "service": config}}}
        with patch.dict(globals(), {"CALLER_SHA": hashlib.sha256(b"offline@example.invalid").hexdigest()}):
            binding = resource_binding("service", value)
            check(audit_binding("service", [event], binding, values)["availableConfigurationMatchesPlan"])
            for field, wrong in (("insertId", "wrong"), ("timestamp", "2026-10-04T22:01:59Z"),
                                 ("receiveTimestamp", "2026-10-04T22:01:58Z")):
                changed = copy.deepcopy(event); changed[field] = wrong
                rejects(lambda changed=changed: audit_binding("service", [changed], binding, values))
            for field, wrong in (("methodName", "google.cloud.run.v1.Services.ReplaceService"), ("status", {"code": 7}),
                                 ("authenticationInfo", {"principalEmail": "other@example.invalid"})):
                changed = copy.deepcopy(event); changed["protoPayload"][field] = wrong
                rejects(lambda changed=changed: audit_binding("service", [changed], binding, values))
            changed = copy.deepcopy(event)
            changed["protoPayload"]["request"]["service"]["spec"]["template"]["spec"]["containers"][0]["image"] = "wrong-image"
            rejects(lambda: audit_binding("service", [changed], binding, values))
            changed = copy.deepcopy(event); changed["protoPayload"]["request"]["extra"] = "unrecognized"
            rejects(lambda: audit_binding("service", [changed], binding, values))
            deleted = copy.deepcopy(event); deleted["protoPayload"]["methodName"] = "google.cloud.run.v1.Services.DeleteService"
            rejects(lambda: audit_binding("service", [event, deleted], binding, values))
            rejects(lambda: audit_binding("service", [event, event], binding, values))
    check(attestation_comparable({"recordedAtUtc": "a", "lifecycleAuditThroughUtc": "b", "exceptions": entries}) == {"exceptions": entries})
    print(json.dumps({"status": "passed", "offlineGuardTests": count, "cloudCalls": 0, "mutationCalls": 0}))
    return 0


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("phase", choices=("inspect", "plan", "delete", "self-test"))
    parser.add_argument("--allow-stage-deletion", action="store_true")
    args = parser.parse_args()
    require(args.allow_stage_deletion == (args.phase == "delete"), "explicit_stage_deletion_flag_required")
    if args.phase == "self-test":
        return self_test()
    if args.phase == "inspect":
        print(json.dumps(collect_attestation(), indent=2))
        return 0
    return run_teardown(args.phase == "delete")


if __name__ == "__main__":
    try:
        raise SystemExit(main())
