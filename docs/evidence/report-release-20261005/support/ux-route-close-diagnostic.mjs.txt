import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';
import { createServer } from 'node:http';
import { chromium } from 'playwright';
import { assertMutationAllowed } from '../../scripts/check-hosted-report-ux-release.mjs';
import { forwardWithoutRedirects, safeFailure } from '../../scripts/check-hosted-onboarding-release.mjs';

// Ignored, local-only reproduction using the exact current newContext closure.
const source = readFileSync(new URL('../../scripts/check-hosted-report-ux-release.mjs', import.meta.url), 'utf8');
const start = source.indexOf('  const newContext = async () => {');
const end = source.indexOf('  const step = ', start);
assert(start > 0 && end > start);
const make = new Function('browser', 'config', 'scope', 'contexts', 'boundaryFailures', 'writes',
  'assertMutationAllowed', 'forwardWithoutRedirects', 'safeFailure', 'requireCondition',
  `let pageErrors = 0;\n${source.slice(start, end)}\nreturn newContext;`);
function requireCondition(ok, code) { if (!ok) { const error = new Error(code); error.safeCode = code; throw error; } }
async function until(read, label) {
  const end = Date.now() + 8000;
  while (Date.now() < end) { if (read()) return; await new Promise((done) => setTimeout(done, 10)); }
  throw new Error(`timeout: ${label}`);
}
let active;
const sockets = new Set();
const server = createServer((req, res) => {
  if (req.url === '/index.html') { res.writeHead(200, { 'content-type': 'text/html' }); res.end('<!doctype html><title>Local guarded-route diagnostic</title>'); return; }
  assert(req.url.startsWith('/api/'));
  active.received.push({ method: req.method, path: req.url });
  active.responses.push(res);
  if (active.serverMode === 'destroy') req.socket.destroy();
});
server.on('connection', (socket) => { sockets.add(socket); socket.on('close', () => sockets.delete(socket)); });
await new Promise((done) => server.listen(0, '127.0.0.1', done));
const baseURL = `http://127.0.0.1:${server.address().port}`;
let browser;
const results = [];
try {
  browser = await chromium.launch({ headless: true });
  for (const kind of ['held_get_close', 'settled_get_close', 'live_get_transport_failure', 'forbidden_post', 'held_allowed_post_close']) {
    for (let run = 1; run <= 3; run += 1) {
      active = { received: [], responses: [], rawErrors: [], terminals: [], serverMode: kind === 'live_get_transport_failure' ? 'destroy' : 'hold' };
      const state = active;
      const contexts = [], boundaryFailures = new Set();
      const scope = { baseURL, worker: {}, supervisor: {}, recoveryTokens: new Set(), uploadPaths: new Set() };
      const forward = async (route, origin) => {
        const path = new URL(route.request().url()).pathname;
        try { await forwardWithoutRedirects(route, origin); }
        catch (error) { state.rawErrors.push({ path, message: String(error.message).split('\n')[0] }); throw error; }
        finally { state.terminals.push(path); }
      };
      const context = await make(browser, { baseURL }, scope, contexts, boundaryFailures, { photos: 0, reports: 0 },
        assertMutationAllowed, forward, safeFailure, requireCondition)();
      const page = await context.newPage();
      await page.goto('/index.html');
      const method = kind.includes('post') ? 'POST' : 'GET';
      const path = kind === 'forbidden_post' ? '/api/form-submissions'
        : kind === 'held_allowed_post_close' ? '/api/auth/logout' : '/api/my-form-submissions?purpose=report';
      await page.evaluate(({ method, path }) => { globalThis.localRequest = fetch(path, { method,
        ...(method === 'POST' ? { headers: { 'content-type': 'application/json' }, body: '{}' } : {}) }).then((r) => r.text()).catch(() => 'failed'); }, { method, path });
      if (kind === 'forbidden_post') {
        await until(() => boundaryFailures.size, 'forbidden POST guard');
        assert.equal(state.received.length, 0);
        assert.deepEqual([...boundaryFailures], ['mutation_outside_owned_fixture_boundary']);
        await context.close();
      } else {
        await until(() => state.received.length > 0, 'request forwarded');
        if (kind === 'settled_get_close') {
          state.responses[0].writeHead(200, { 'content-type': 'application/json' }); state.responses[0].end('[]');
          await page.evaluate(() => globalThis.localRequest);
          await until(() => state.terminals.some((item) => item.startsWith('/api/')), 'completed guarded forwarding');
          await context.close();
          assert.equal(boundaryFailures.size, 0);
        } else if (kind === 'live_get_transport_failure') {
          await until(() => boundaryFailures.size, 'live network failure');
          assert.deepEqual([...boundaryFailures], ['browser_or_api_operation_failed']);
          assert.equal(page.isClosed(), false);
          await context.close();
        } else {
          assert.equal(boundaryFailures.size, 0);
          await context.close();
          await until(() => boundaryFailures.size, 'close cancellation');
          assert.deepEqual([...boundaryFailures], ['browser_or_api_operation_failed']);
        }
      }
      for (const response of state.responses) response.destroy();
      results.push({ kind, run, received: state.received, failures: [...boundaryFailures], rawErrors: state.rawErrors });
    }
  }
  console.log(JSON.stringify({ status: 'passed', exactGuardedContextSource: true, loopbackOnly: true, results }, null, 2));
} finally {
  await browser?.close();
  for (const socket of sockets) socket.destroy();
  await new Promise((done) => server.close(done));
}
